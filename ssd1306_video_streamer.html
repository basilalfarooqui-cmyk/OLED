// ESP32 High-Speed SSD1306 WebSocket Receiver
// Requires: WebSockets by Markus Sattler & U8g2 by olikraus

#include <WiFi.h>
#include <WebSocketsServer.h>
#include <Wire.h>
#include <U8g2lib.h>

// Display constructor for SSD1306 (Hardware I2C)
U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, U8X8_PIN_NONE);

// Set your WiFi credentials (or keep as Access Point)
const char* ssid = "OLED_Streamer";
const char* password = "password123";

WebSocketsServer webSocket = WebSocketsServer(81);

void onWebSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
  if (type == WStype_BIN) {
    if (length == 1024) { // Exactly 1 frame
      uint8_t* buf = u8g2.getBufferPtr();
      memcpy(buf, payload, 1024);
      u8g2.sendBuffer();
    }
  } else if (type == WStype_CONNECTED) {
    Serial.printf("[%u] Connected!\n", num);
  } else if (type == WStype_DISCONNECTED) {
    Serial.printf("[%u] Disconnected!\n", num);
  }
}

void setup() {
  Serial.begin(115200);

  // Initialize display
  u8g2.begin();
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_6x10_tf);
  u8g2.drawStr(10, 35, "WiFi Waiting...");
  u8g2.sendBuffer();

  // Overclock I2C bus to 800 kHz for 30+ FPS
  Wire.setClock(800000);

  // Start ESP32 as SoftAP
  WiFi.softAP(ssid, password);
  Serial.print("Access Point IP: ");
  Serial.println(WiFi.softAPIP());

  webSocket.begin();
  webSocket.onEvent(onWebSocketEvent);
}

void loop() {
  webSocket.loop();
}
